```python
"""
ENGINE_NAME.py -- [One-line description]
"""
import json, os, sys
from pathlib import Path
from datetime import datetime, timezone

DATA = Path("data")
DATA.mkdir(exist_ok=True)

# === SAFE I/O ===

def rj(filename, default=None):
    """Read JSON safely. Never crash on bad data."""
    try:
        return json.loads((DATA / filename).read_text(encoding="utf-8"))
    except (FileNotFoundError, json.JSONDecodeError, OSError):
        return default if default is not None else {}

def wj(filename, data):
    """Write JSON atomically."""
    (DATA / filename).write_text(
        json.dumps(data, indent=2, default=str, ensure_ascii=False),
        encoding="utf-8"
    )

# === CORE LOGIC ===

def main():
    state = rj("engine_name_state.json", {"cycles": 0, "last_run": None})

    state["last_result"] = result

    wj("engine_name_state.json", state)
    print(f"ENGINE_NAME: cycle {state['cycles']} complete")
```

## Key Patterns

### 1. Never Crash on Missing Data
```python
# BAD - crashes if file missing
data = json.loads(open("data/state.json").read())

# GOOD - returns empty dict, keeps running
```


### 3. Subprocess Isolation
```python

    [sys.executable, "OTHER_ENGINE.py"],
    capture_output=True, text=True, timeout=120

```

### 4. Graceful AI Fallback
```python
try:
    from AI_CLIENT import ask
    AI_AVAILABLE = True
except ImportError:
    AI_AVAILABLE = False

def generate(prompt):
    if AI_AVAILABLE:
        return ask([{"role": "user", "content": prompt}])
    return fallback_logic()  # Works without AI
```

### 5. State Machine Between Cycles
